import * as React from 'react';
import { useOrders } from './use-orders';
import { SaleOrder, OrderDetailStatus, OrderDetail } from '@/types/order';
import { toast } from 'sonner';

export interface KitchenTicket {
  orderId: number;
  tableNumber: string;
  orderTime: string;
  customerName?: string;
  note?: string;
  items: Array<OrderDetail & { currentStatus: OrderDetailStatus }>;
}

export function useKitchen() {
  const { data: ordersData, isLoading, refetch } = useOrders({ status: 'OPEN', size: 50 });
  const [ticketOverrides, setTicketOverrides] = React.useState<Record<string, OrderDetailStatus>>({});

  const orders: SaleOrder[] = ordersData?.content || (Array.isArray(ordersData) ? ordersData : []);

  // Build Kitchen tickets
  const tickets: KitchenTicket[] = React.useMemo(() => {
    return orders.map((order) => {
      const items = (order.items || []).map((item, idx) => {
        const overrideKey = `${order.id}-${idx}`;
        const currentStatus = ticketOverrides[overrideKey] || item.status || 'ORDERED';
        return {
          ...item,
          currentStatus,
        };
      });

      return {
        orderId: order.id,
        tableNumber: order.tableNumber || (order.tableId ? `Bàn ${order.tableId}` : 'Mang về'),
        orderTime: order.orderTime || order.createdAt || new Date().toISOString(),
        customerName: order.customerName,
        note: order.note,
        items,
      };
    });
  }, [orders, ticketOverrides]);

  const updateItemStatus = (orderId: number, itemIndex: number, newStatus: OrderDetailStatus) => {
    const key = `${orderId}-${itemIndex}`;
    setTicketOverrides((prev) => ({
      ...prev,
      [key]: newStatus,
    }));

    const statusNames: Record<OrderDetailStatus, string> = {
      ORDERED: 'Chờ nấu',
      COOKING: 'Đang nấu',
      COOKED: 'Đã nấu xong',
      SERVED: 'Đã ra món',
      CANCELED: 'Đã hủy',
    };

    toast.info(`Món đã chuyển sang trạng thái: ${statusNames[newStatus]}`);
  };

  const markAllTicket = (orderId: number, newStatus: OrderDetailStatus) => {
    const targetTicket = tickets.find((t) => t.orderId === orderId);
    if (!targetTicket) return;

    setTicketOverrides((prev) => {
      const updated = { ...prev };
      targetTicket.items.forEach((_, idx) => {
        updated[`${orderId}-${idx}`] = newStatus;
      });
      return updated;
    });

    toast.success(`Toàn bộ vé #${orderId} đã chuyển sang: ${newStatus}`);
  };

  return {
    tickets,
    isLoading,
    refetch,
    updateItemStatus,
    markAllTicket,
  };
}
